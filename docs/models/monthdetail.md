# MonthDetail

## Example Usage

```typescript
import { MonthDetail } from "ynab-ts/models";
import { RFCDate } from "ynab-ts/types";

let value: MonthDetail = {
  month: new RFCDate("2026-04-22"),
  income: 577129,
  budgeted: 163068,
  activity: 651562,
  toBeBudgeted: 765286,
  deleted: true,
  categories: [],
};
```
